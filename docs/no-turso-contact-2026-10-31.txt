4. Turso CLI invocation inventory by date
-----------------------------------------
The session transcript is JSONL; executed shell commands are in `command`
fields. This inventory lists each distinct command form once (repeated runs are
collapsed); loops are expanded to their literal subcommand sets. Quoted text
and grep patterns are not executions. Commands that invoke `turso` inside a
help-capture loop are executions and are included.

To re-derive the counts, split each bash command on shell separators **that are
outside quotes**, then keep the segments whose first word is `turso` (optionally
after `timeout N `). Splitting naively on `|` is wrong: a grep pattern such as
`grep -n 'A\|B\|turso auth whoami'` puts pipes inside a quoted string, and
splitting there fabricates commands that never ran. The script below is the
check this list was produced from:

  import collections, json, re, sys
  from pathlib import Path

  def shell_segments(command):
      """Split on separators outside quotes; honour backslash escapes in "..."."""
      out, buf, i, quote = [], [], 0, None
      while i < len(command):
          ch = command[i]
          if quote == "'":
              buf.append(ch)
              if ch == "'":
                  quote = None
              i += 1
              continue
          if quote == '"':
              if ch == "\\" and i + 1 < len(command):
                  buf.append(command[i : i + 2]); i += 2; continue
              buf.append(ch)
              if ch == '"':
                  quote = None
              i += 1
              continue
          if ch in ("'", '"'):
              quote = ch; buf.append(ch); i += 1; continue
          if ch == "\\" and i + 1 < len(command):
              buf.append(command[i : i + 2]); i += 2; continue
          joined = False
          for sep in ("&&", "||"):
              if command.startswith(sep, i):
                  out.append("".join(buf)); buf = []; i += len(sep); joined = True; break
          if joined:
              continue
          if ch in (";", "\n", "|"):
              out.append("".join(buf)); buf = []; i += 1
          else:
              buf.append(ch); i += 1
      out.append("".join(buf))
      for segment in out:
          segment = re.sub(r"\s*2>&1", " ", segment).strip()
          segment = re.sub(r"^timeout\s+\d+\s+", "", segment).strip()
          if segment:
              yield segment

  forms = collections.defaultdict(collections.Counter)
  for line in Path(sys.argv[1]).read_text(errors="replace").splitlines():
      try:
          record = json.loads(line)
      except ValueError:
          continue
      message = record.get("message")
      if not isinstance(message, dict):
          continue
      for item in message.get("content", []):
          if item.get("type") != "toolCall" or item.get("name") != "bash":
              continue
          command = (item.get("arguments") or {}).get("command", "")
          if not isinstance(command, str):
              continue
          if re.search(r"(?:python3?|cat)\s+[^\n]*<<", command):
              continue  # a program body, not a shell command
          for segment in shell_segments(command):
              if segment.startswith("turso "):
                  forms[record["timestamp"][:10]][" ".join(segment.split()[:2])] += 1

  for day in sorted(forms):
      print(day, dict(forms[day]))
